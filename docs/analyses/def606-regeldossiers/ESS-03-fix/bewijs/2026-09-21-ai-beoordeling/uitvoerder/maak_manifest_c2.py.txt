"""Manifest correctieronde 2 (DEF-766): hashes van ALLE gewijzigde/nieuwe bestanden in
de werkboom t.o.v. base (met vergelijking t.o.v. correctiemanifest v1), actuele
prompt/norm/modelhash (ongewijzigd t.o.v. ronde 1: geen nieuwe modelcalls), de
transportproef (C) en de exitcodes van ronde 2. Schrijft correction-manifest-v2.json."""

from __future__ import annotations

import hashlib
import json
import subprocess
import sys
from datetime import UTC, datetime
from pathlib import Path

WERKBOOM = Path("/Users/chrislehnen/.codex/worktrees/2075/Definitie-app")
UIT = Path("/tmp/def766-ai-20260921")
BASE = "2c9a6e3a13afa4ecbe39163cc418e2b0f8c41644"
sys.path.insert(0, str(WERKBOOM / "src"))


def sha(pad: Path) -> str:
    return hashlib.sha256(pad.read_bytes()).hexdigest()


def git(*args: str) -> str:
    return subprocess.check_output(["git", *args], cwd=WERKBOOM, text=True).strip()


def main() -> None:
    from services.validation.ess03_assessment_service import (
        Ess03AssessmentService,
        bouw_beoordelingsprompt,
        laad_ess03_norm,
    )

    norm = laad_ess03_norm()
    system_prompt, _ = bouw_beoordelingsprompt(
        "x", "y", {}, (), intentie=None, norm=norm
    )
    norm_sha = hashlib.sha256(
        json.dumps(norm, ensure_ascii=False, sort_keys=True).encode("utf-8")
    ).hexdigest()

    v1 = json.loads((UIT / "correction-manifest-v1.json").read_text(encoding="utf-8"))
    v1_hashes = {pad: info["sha256"] for pad, info in v1["bestanden"].items()}
    seal = json.loads(
        (UIT / "browser-v2-input-and-source-seal.json").read_text(encoding="utf-8")
    )["files"]

    status = subprocess.check_output(
        ["git", "status", "--porcelain", "--untracked-files=all"],
        cwd=WERKBOOM,
        text=True,
    )
    bestanden: dict[str, dict] = {}
    for regel in status.splitlines():
        if len(regel) < 4:
            continue
        code, pad = regel[:2], regel[3:]
        if pad.startswith("docs/adr/ADR-002") or pad.startswith("docs/plans/"):
            continue  # niet van de uitvoerder
        if "__pycache__" in pad or pad.startswith(".claude/"):
            continue
        p = WERKBOOM / pad
        if p.is_file():
            h = sha(p)
            eerder = v1_hashes.get(pad)
            info = {
                "git_status": code.strip() or "M",
                "sha256": h,
                "t_o_v_correctie_v1": (
                    "nieuw in ronde 2"
                    if eerder is None
                    else ("gewijzigd in ronde 2" if eerder != h else "ongewijzigd")
                ),
            }
            if pad in seal:
                info["browser_v2_seal_sha256"] = seal[pad]
                info["gelijk_aan_browser_v2_seal"] = seal[pad] == h
            bestanden[pad] = info

    transport = json.loads(
        (UIT / "c2-c-transportproef.json").read_text(encoding="utf-8")
    )
    exitcodes = json.loads(
        (UIT / "exitcodes-correctie-v2.json").read_text(encoding="utf-8")
    )
    run_v2 = json.loads((UIT / "ontwikkelrun-v2.json").read_text(encoding="utf-8"))

    manifest = {
        "schema": "def766-correctie-manifest/2",
        "gemaakt": datetime.now(UTC).isoformat(),
        "uitvoerder": (
            "Claude Code CLI (Opus 5, 1M context) — correctieronde 2 op restpunten "
            "A (R1/R5), B (R6), C (onderzoek verbindingsfout); geen commits/push/PR/"
            "deletes; geen nieuwe modelcalls"
        ),
        "werkboom": str(WERKBOOM),
        "branch": git("branch", "--show-current"),
        "base": BASE,
        "head_commit_bij_manifest": git("rev-parse", "HEAD"),
        "opmerking_head": (
            "geen nieuwe commits door de uitvoerder; alle wijzigingen (fase 1 + "
            "correctieronden 1 en 2) staan ongecommit in de werkboom"
        ),
        "prompt": {
            "prompt_version": Ess03AssessmentService.PROMPT_VERSION,
            "system_prompt_sha256": hashlib.sha256(
                system_prompt.encode("utf-8")
            ).hexdigest(),
            "norm_sha256": norm_sha,
            "ongewijzigd_t_o_v_correctie_v1": (
                v1["prompt"]["system_prompt_sha256"]
                == hashlib.sha256(system_prompt.encode("utf-8")).hexdigest()
                and v1["prompt"]["norm_sha256"] == norm_sha
                and v1["prompt"]["prompt_version"]
                == Ess03AssessmentService.PROMPT_VERSION
            ),
            "contract_version": "ess03/1",
        },
        "modelconfig": run_v2["modelconfig"],
        "modelcalls_ronde_2": 0,
        "ontwikkelrun_v2_behouden": {
            "bestand": str(UIT / "ontwikkelrun-v2.json"),
            "sha256": sha(UIT / "ontwikkelrun-v2.json"),
            "correct": f"{run_v2['correct']}/{run_v2['aantal']}",
        },
        "transportproef_c": {
            "script": "scripts/ess03/proef_eventloop_transport.py",
            "bestand": str(UIT / "c2-c-transportproef.json"),
            "sha256": sha(UIT / "c2-c-transportproef.json"),
            "log": str(UIT / "c2-c-transportproef.log"),
            "alle_conform": transport["alle_conform"],
            "proeven": [
                {"naam": p["naam"], "conform": p["conform"]} for p in transport["proeven"]
            ],
            "requests_ontvangen": transport["requests_ontvangen"],
        },
        "budget": {
            "totaal_transportpogingen_conservatief_na_browser_v2": 26,
            "nieuwe_echte_modelcalls_ronde_2": 0,
            "geplande_eindtests": 30,
            "plafond": 60,
            "ruimte_na_eindtests": 4,
        },
        "exitcodes": exitcodes,
        "bestanden": bestanden,
    }
    (UIT / "correction-manifest-v2.json").write_text(
        json.dumps(manifest, ensure_ascii=False, indent=2), encoding="utf-8"
    )
    print(json.dumps(manifest["prompt"], indent=2))
    print(len(bestanden), "bestanden in manifest")


if __name__ == "__main__":
    main()
