"""Manifest fase 1 (DEF-766): hashes van alle gewijzigde/nieuwe software- en testbestanden,
de gebruikte beoordelingsprompt (systeemprompt + normhash), modelconfig, exitcodes en de
12 modeluitkomsten. Draaien vanuit de werkboom; geen sleutels."""

from __future__ import annotations

import hashlib
import json
import subprocess
import sys
from datetime import UTC, datetime
from pathlib import Path

WERKBOOM = Path("/Users/chrislehnen/.codex/worktrees/2075/Definitie-app")
UIT = Path("/tmp/def766-ai-20260921")
sys.path.insert(0, str(WERKBOOM / "src"))


def sha(pad: Path) -> str:
    return hashlib.sha256(pad.read_bytes()).hexdigest()


def git(*args: str) -> str:
    return subprocess.check_output(["git", *args], cwd=WERKBOOM, text=True).strip()


def main() -> None:
    from services.validation.ess03_assessment_service import (
        Ess03AssessmentService,
        bouw_beoordelingsprompt,
        laad_ess03_norm,
    )

    norm = laad_ess03_norm()
    system_prompt, _ = bouw_beoordelingsprompt(
        "x", "y", {}, (), intentie=None, norm=norm
    )
    norm_sha = hashlib.sha256(
        json.dumps(norm, ensure_ascii=False, sort_keys=True).encode("utf-8")
    ).hexdigest()

    status = subprocess.check_output(
        ["git", "status", "--porcelain", "--untracked-files=all"], cwd=WERKBOOM, text=True
    )
    bestanden: dict[str, dict] = {}
    for regel in status.splitlines():
        if len(regel) < 4:
            continue
        code, pad = regel[:2], regel[3:]
        if pad.startswith("docs/adr/ADR-002") or pad.startswith("docs/plans/"):
            continue  # niet van de uitvoerder
        p = WERKBOOM / pad
        if p.is_file():
            bestanden[pad] = {"git_status": code.strip() or "M", "sha256": sha(p)}

    run = json.loads((UIT / "ontwikkelrun-v1.json").read_text(encoding="utf-8"))
    uitkomsten = [
        {
            "id": r["id"],
            "verwacht": r["verwacht"],
            "gekregen": r["gekregen_verdict"],
            "correct": r["correct"],
            "status_deel": r["gekregen_status"],
            "duur_s": (r.get("aanroep") or {}).get("duur_s"),
            "tokens_used": (r.get("aanroep") or {}).get("tokens_used"),
            "rejected": len(r["beoordeling"].get("rejected") or []),
            "raw_response_sha256": r["hashes"]["raw_response_sha256"],
            "user_prompt_sha256": r["hashes"]["user_prompt_sha256"],
            "fingerprint": r["hashes"]["fingerprint"],
        }
        for r in run["resultaten"]
    ]

    exitcodes = json.loads((UIT / "exitcodes-fase1.json").read_text(encoding="utf-8"))

    manifest = {
        "schema": "def766-fase1-manifest/1",
        "gemaakt": datetime.now(UTC).isoformat(),
        "uitvoerder": "Claude Code CLI (Opus 5, 1M context) — implementatie; geen commits/push/PR",
        "werkboom": str(WERKBOOM),
        "branch": git("branch", "--show-current"),
        "base": "2c9a6e3a13afa4ecbe39163cc418e2b0f8c41644",
        "head_commit_bij_manifest": git("rev-parse", "HEAD"),
        "opmerking_head": "geen nieuwe commits door de uitvoerder; alle wijzigingen staan ongecommit in de werkboom (zie bestanden)",
        "prompt": {
            "prompt_version": Ess03AssessmentService.PROMPT_VERSION,
            "system_prompt_sha256": hashlib.sha256(system_prompt.encode("utf-8")).hexdigest(),
            "norm_sha256": norm_sha,
            "norm_bron": "src/toetsregels/regels/ESS-03.json (uitleg, toelichting, toetsvraag, geldigheid)",
            "prompt_builder": "src/services/validation/ess03_assessment_service.py::bouw_beoordelingsprompt",
            "contract_version": "ess03/1",
        },
        "modelconfig": run["modelconfig"],
        "ontwikkelrun": {
            "bestand": str(UIT / "ontwikkelrun-v1.json"),
            "sha256": sha(UIT / "ontwikkelrun-v1.json"),
            "gevallenbestand": run["gevallenbestand"],
            "aantal_calls": run["aantal"],
            "correct": run["correct"],
            "per_klasse": run["per_klasse"],
            "uitkomsten": uitkomsten,
        },
        "exitcodes": exitcodes,
        "bestanden": bestanden,
    }
    (UIT / "phase1-manifest-v1.json").write_text(
        json.dumps(manifest, ensure_ascii=False, indent=2), encoding="utf-8"
    )
    print(json.dumps({k: manifest[k] for k in ("prompt", "branch", "head_commit_bij_manifest")}, indent=2))
    print(len(bestanden), "bestanden in manifest")


if __name__ == "__main__":
    main()
