OP_ID = {g["id"]: g for g in BASIS}


def variant(basis_id: str, suffix: str, titel: str, doel: str) -> dict:
    v = copy.deepcopy(OP_ID[basis_id])
    v["id"] = f"{basis_id}-{suffix}"
    v["basis"] = basis_id
    v["titel"] = titel
    v["doel"] = doel
    return v
