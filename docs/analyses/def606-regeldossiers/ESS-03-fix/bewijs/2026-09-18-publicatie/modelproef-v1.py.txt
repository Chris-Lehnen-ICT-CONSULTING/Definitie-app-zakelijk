# Alleen signaalindicatoren voor de handmatige analyse — geen automatisch oordeel.
IDENTIFIER_RE = re.compile(
    r"\b(nummer|code|registratie\w*|identificat\w*|isbn|serienummer|kenteken|"
    r"barcode|label|uniek\w*|geregistreerd\w*)\b",
    re.IGNORECASE,
)
MONSTER_RE = re.compile(
    r"\b(monster\w*|portie\w*|partij\w*|hoeveelheid|bemonster\w*|afgenomen)\b",
    re.IGNORECASE,
)
PII_RE = re.compile(
    r"[\w.+-]+@[\w-]+\.[\w.]+"  # e-mail
    r"|\b(\+31|0031|06)[\s-]?\d[\d\s-]{7,}\b"  # telefoon
    r"|\b\d{9}\b"  # BSN-achtig
    r"|clictconsulting|lehnen",  # gebruikersidentiteit hoort niet in een prompt
    re.IGNORECASE,
)


def _sha(text: str) -> str:
    return hashlib.sha256(text.encode("utf-8")).hexdigest()


async def run() -> dict:
    from config.config_manager import get_config_manager, get_prompt_temperature
    from opschoning.opschoning_enhanced import extract_definition_from_gpt_response
    from services.ai import create_ai_client
    from services.ai.model_router import ModelRouter
    from services.ai_service_v2 import AIServiceV2
    from services.interfaces import GenerationRequest
    from services.modelantwoord import CONFLICT_SENTINEL, lees_modelantwoord
    from services.prompts.prompt_service_v2 import PromptServiceV2
    from services.validation.modular_validation_service import (
        ModularValidationService,
    )
    from toetsregels.manager import get_toetsregel_manager
    from toetsregels.rule_cache import get_rule_cache
    from utils.async_api import RateLimitConfig

    cfg = get_config_manager()
    provider = cfg.api.ai_provider
    api_key = (
        cfg.api.anthropic_api_key if provider == "anthropic" else cfg.api.openai_api_key
    )
    router = ModelRouter.from_config()
    routed_provider, routed_model = router.get_model("definition_core")
    temperature = get_prompt_temperature("definition")

    verslag: dict = {
        "datum": time.strftime("%Y-%m-%dT%H:%M:%S%z"),
        "python": sys.version.split()[0],
        "werkboom": str(ROOT),
        "env_geladen_uit_project_dotenv": _ENV_GELADEN,
        "provider_geconfigureerd": provider,
        "model_router_definition_core": {"provider": routed_provider, "model": routed_model},
        "api_key_aanwezig": bool(api_key),
        "temperature_definition": temperature,
        "max_tokens": MAX_TOKENS,
        "timeout_s": TIMEOUT_S,
        "retries": {"client_lus_max_retries": 1, "AI_SDK_MAX_RETRIES": os.environ["AI_SDK_MAX_RETRIES"]},
        "use_cache": False,
        "aanroepen_gedaan": 0,
        "cases": [],
    }
    if not api_key:
        verslag["fout"] = (
            f"Geen API-key voor provider {provider!r} beschikbaar via .env/omgeving; "
            "proef niet uitgevoerd (geen mock ingezet)."
        )
        return verslag
    if routed_provider != provider:
        verslag["waarschuwing"] = (
            f"ModelRouter active_provider {routed_provider!r} != config {provider!r}"
        )

    # Échte keten, gespiegeld aan ServiceContainer.ai_service(): zelfde factory,
    # zelfde router; alleen cache uit en de retry-lus op één poging.
    client = create_ai_client(provider=provider, api_key=api_key)
    ai = AIServiceV2(
        rate_limit_config=RateLimitConfig(max_retries=1),
        use_cache=False,
        ai_client=client,
        model_router=router,
    )
    prompts = PromptServiceV2()
    get_rule_cache().clear_cache()
    validator = ModularValidationService(get_toetsregel_manager(), None, None)

    for case in CASES:
        rij: dict = {"id": case["id"], "invoer": {k: v for k, v in case.items() if k != "verwachte_grenzen"},
                     "verwachte_grenzen": case["verwachte_grenzen"]}
        request = GenerationRequest(
            id=f"def766-modelproef-{case['id']}",
            begrip=case["begrip"],
            ontologische_categorie=case["ontologische_categorie"],
            organisatorische_context=case["organisatorische_context"],
            betekenisverduidelijking=case["betekenisverduidelijking"],
            actor="def766-modelproef",
        )
        prompt_result = await prompts.build_generation_prompt(request)
        prompt = prompt_result.text
        (UIT / f"modelproef-prompt-{case['id']}.txt").write_text(prompt, encoding="utf-8")
        rij["prompt"] = {
            "sha256": _sha(prompt),
            "tekens": len(prompt),
            "ess03_kaart_aantal": prompt.count("🔹 **ESS-03"),
            "ess03_g_kern_aanwezig": "verzin geen nummer, bron of telconventie" in prompt,
            "oude_nummerinstructie_aanwezig": "Noem criteria voor unieke identificatie" in prompt,
            "verduidelijking_in_prompt": case["betekenisverduidelijking"] in prompt,
            "pii_treffers": sorted({m.group(0) for m in PII_RE.finditer(prompt)}),
            "bronnenblok_aanwezig": "<bronnen>" in prompt,
            "metadata_source_receipt": prompt_result.metadata.get("source_receipt"),
        }
        if rij["prompt"]["pii_treffers"]:
            rij["resultaat"] = {"status": "overgeslagen", "reden": "PII-treffer in uitgaande prompt"}
            verslag["cases"].append(rij)
            continue
        if verslag["aanroepen_gedaan"] >= MAX_CALLS:
            rij["resultaat"] = {"status": "overgeslagen", "reden": "plafond van 4 aanroepen bereikt"}
            verslag["cases"].append(rij)
            continue
        if os.environ.get("MODELPROEF_DRY") == "1":
            rij["resultaat"] = {"status": "dry_run", "reden": "alleen promptopbouw, geen modelaanroep"}
            verslag["cases"].append(rij)
            continue

        verslag["aanroepen_gedaan"] += 1
        t0 = time.time()
        try:
            res = await ai.generate_definition(
                prompt=prompt,
                temperature=temperature,
                max_tokens=MAX_TOKENS,
                timeout_seconds=TIMEOUT_S,
            )
        except Exception as exc:  # technische fout is een bevinding, geen retry
            rij["resultaat"] = {
                "status": "technische_fout",
                "fouttype": type(exc).__name__,
                "melding": re.sub(r"sk-[A-Za-z0-9_-]+", "sk-***", str(exc))[:500],
                "duur_s": round(time.time() - t0, 2),
            }
            verslag["cases"].append(rij)
            continue

        raw = res.text
        antwoord = lees_modelantwoord(raw)
        conflict = raw.strip().startswith(CONFLICT_SENTINEL)
        kandidaat = "" if conflict else extract_definition_from_gpt_response(raw)
        rij["resultaat"] = {
            "status": "conflictmelding" if conflict else "definitie",
            "model": res.model,
            "cached": res.cached,
            "tokens_used_geschat": res.tokens_used,
            "tokens_estimated": bool(res.metadata.get("tokens_estimated")),
            "generation_time_s": round(res.generation_time, 2),
            "retry_count": res.retry_count,
            "raw_text": raw,
            "raw_sha256": _sha(raw),
            "modelantwoord_soort": type(antwoord).__name__,
            "modelantwoord_geldig": getattr(antwoord, "geldig", None),
            "kandidaat_geextraheerd": kandidaat,
        }
        rij["signalen"] = {
            "identifier_woorden_in_kandidaat": sorted({m.group(0).lower() for m in IDENTIFIER_RE.finditer(kandidaat)}),
            "monster_woorden_in_kandidaat": sorted({m.group(0).lower() for m in MONSTER_RE.finditer(kandidaat)}),
            "verduidelijking_kernwoorden_terug": [
                w for w in ("aaneengesloten", "peilmoment", "stof", "vloeistof", "bemonstering", "exemplaar", "ISBN")
                if w.lower() in kandidaat.lower()
            ],
        }
        if kandidaat:
            val = await validator.validate_definition(
                begrip=case["begrip"],
                text=kandidaat,
                ontologische_categorie=case["ontologische_categorie"],
                context={"organisatorische_context": case["organisatorische_context"]},
            )
            item = next((r for r in val["review_required"] if r["rule_id"] == "ESS-03"), None)
            rij["toetsing_kandidaat"] = {
                "validation_status": val["validation_status"],
                "ess03_status": val["rule_statuses"].get("ESS-03"),
                "ess03_violation": any(v.get("code") == "ESS-03" for v in val["violations"]),
                "ess03_signals": (item or {}).get("signals"),
                "ess03_reden": (item or {}).get("reason"),
                "overall_score": val["overall_score"],
                "gefaalde_regels": sorted(v["code"] for v in val["violations"]),
            }
        verslag["cases"].append(rij)

    return verslag

              f"ess03_kaart={c['prompt']['ess03_kaart_aantal']} pii={c['prompt']['pii_treffers']}")
